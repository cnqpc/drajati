// تصفية حسب نص البحث const matchSearch = !searchVal || (post.title && post.title.toLowerCase().includes(searchVal)) || (post.description && post.description.toLowerCase().includes(searchVal)) || (post.price && post.price.toString().includes(searchVal)); return matchCity && matchCategory && matchSearch; }); renderPostsFeed(filteredPosts); } // 10. عرض المنشورات وتعبئة القالب الديناميكي (Render Posts UI) function renderPostsFeed(posts) { const feed = document.getElementById('posts-feed'); const noResults = document.getElementById('no-results'); const template = document.getElementById('post-card-template'); if (!feed || !template) return; feed.innerHTML = ''; if (!posts || posts.length === 0) { if (noResults) noResults.classList.remove('hidden'); return; } if (noResults) noResults.classList.add('hidden'); posts.forEach(post => { const clone = template.content.cloneNode(true); const card = clone.querySelector('.post-card'); // تعبئة بيانات الناشر const avatar = clone.querySelector('.user-avatar'); avatar.src = post.user_avatar || 'https://api.dicebear.com/7.x/bottts/svg?seed=moped'; clone.querySelector('.user-name').textContent = post.user_name || 'مستخدم دراجتي'; clone.querySelector('.post-city').textContent = post.city || 'بغداد'; clone.querySelector('.time-ago').textContent = formatTimeAgo(post.created_at); clone.querySelector('.post-condition').textContent = post.condition || 'مستعمل'; // تعبئة تفاصيل الدراجة const img = clone.querySelector('.post-image'); img.src = (post.images && post.images.length > 0) ? post.images[0] : 'https://images.unsplash.com/photo-1558981806-ec527fa84c39?auto=format&fit=crop&w=600&q=80'; clone.querySelector('.post-price').textContent = `${post.price} ${post.currency || '$'}`; clone.querySelector('.post-title').textContent = post.title; clone.querySelector('.post-desc').textContent = post.description || ''; // أعداد التفاعلات const likesCount = clone.querySelector('.likes-count'); likesCount.textContent = post.likes_count || 0; const likeBtn = clone.querySelector('.like-btn'); const isLiked = post.likes_users && currentUser && post.likes_users.includes(currentUser.id); if (isLiked) { likeBtn.classList.add('text-red-500'); likeBtn.querySelector('i').className = 'fa-solid fa-heart text-sm text-red-500'; } likeBtn.onclick = () => handleLikePost(post.id, isLiked); // زر المفضلة const saveBtn = clone.querySelector('.save-btn'); const isSaved = savedPostIds.has(post.id); if (isSaved) { saveBtn.classList.add('text-amber-500'); saveBtn.querySelector('i').className = 'fa-solid fa-bookmark text-sm text-amber-500'; } saveBtn.onclick = () => handleToggleSavePost(post.id); // أزرار الاتصال والمحادثة const commentBtn = clone.querySelector('.comment-btn'); commentBtn.onclick = () => openCommentsModal(post.id); const whatsappBtn = clone.querySelector('.whatsapp-btn'); if (post.whatsapp) { whatsappBtn.href = `https://wa.me/${post.whatsapp.replace(/[^0-9]/g, '')}?text=${encodeURIComponent('السلام عليكم، بخصوص إعلانك على دراجتي: ' + post.title)}`; whatsappBtn.classList.remove('hidden'); } feed.appendChild(clone); }); } // 11. أزرار تصفية المحافظة والفئات function selectCity(city) { currentCityFilter = city; document.getElementById('active-city-label').textContent = city === 'الكل' ? 'جميع المحافظات' : city; document.querySelectorAll('.city-btn').forEach(btn => { if (btn.textContent.includes(city)) { btn.className = 'city-btn active bg-brand-500 text-white px-3 py-1.5 rounded-xl whitespace-nowrap shadow-sm btn-tap'; } else { btn.className = 'city-btn bg-white dark:bg-gray-800 text-gray-700 dark:text-gray-300 px-3 py-1.5 rounded-xl whitespace-nowrap border dark:border-gray-700 btn-tap'; } }); applyFilters(); } function filterByCategory(category) { currentCategoryFilter = category; document.getElementById('selected-category-label').textContent = category === 'الكل' ? 'الفئات' : category; toggleModal('categories-modal'); applyFilters(); } function filterPosts() { const searchVal = document.getElementById('search-input').value; const clearBtn = document.getElementById('clear-search-btn'); if (searchVal.length > 0) { clearBtn.classList.remove('hidden'); } else { clearBtn.classList.add('hidden'); } applyFilters(); } function clearSearch() { document.getElementById('search-input').value = ''; document.getElementById('clear-search-btn').classList.add('hidden'); applyFilters(); } function resetFilters() { currentCityFilter = 'الكل'; currentCategoryFilter = 'الكل'; document.getElementById('search-input').value = ''; document.getElementById('active-city-label').textContent = 'جميع المحافظات'; document.getElementById('selected-category-label').textContent = 'الفئات'; applyFilters(); } // 12. دالة تنسيق الوقت العراقي البسيط (Time Format Helper) function formatTimeAgo(dateString) { if (!dateString) return 'الآن'; const date = new Date(dateString); const now = new Date(); const diffSec = Math.floor((now - date) / 1000); if (diffSec < 60) return 'قبل لحظات'; if (diffSec < 3600) return `قبل ${Math.floor(diffSec / 60)} دقيقة`; try { const { data, error } = await supabaseClient .from('posts') .insert([newPost]) .select(); if (error) throw error; toggleModal('add-post-modal'); e.target.reset(); await fetchPosts(); alert('تم نشر إعلانك بنجاح!'); } catch (err) { console.error('خطأ أثناء إضافة الإعلان:', err); alert('حدث خطأ أثناء حفظ الإعلان، يرجى المحاولة لاحقاً.'); } finally { btn.disabled = false; btn.textContent = 'نشر الإعلان الآن'; } } // 14. التفاعل بالإعجاب (Like/Unlike) async function handleLikePost(postId, isLiked) { if (!currentUser) return; const post = allPosts.find(p => p.id === postId); if (!post) return; let updatedLikes = post.likes_users || []; if (isLiked) { updatedLikes = updatedLikes.filter(id => id !== currentUser.id); } else { updatedLikes.push(currentUser.id); } const newCount = updatedLikes.length; // تحديث الواجهة فورياً قبل رد السيرفر post.likes_users = updatedLikes; post.likes_count = newCount; renderPostsFeed(filteredPosts); try { await supabaseClient .from('posts') .update({ likes_count: newCount, likes_users: updatedLikes }) .eq('id', postId); } catch (err) { console.error('خطأ في الإعجاب:', err); } } // 15. حفظ الإعلانات في قائمة المفضلة async function handleToggleSavePost(postId) { if (!currentUser) return; if (savedPostIds.has(postId)) { savedPostIds.delete(postId); } else { savedPostIds.add(postId); } renderPostsFeed(filteredPosts); try { await supabaseClient .from('saved_posts') .upsert({ user_id: currentUser.id, post_id: postId, saved_at: new Date() }); } catch (err) { console.error('خطأ أثناء الحفظ للمفضلة:', err); } } async function fetchSavedPostIds() { if (!currentUser) return; try { const { data } = await supabaseClient .from('saved_posts') .select('post_id') .eq('user_id', currentUser.id); if (data) { savedPostIds = new Set(data.map(item => item.post_id)); } } catch (err) { console.error('خطأ في جلب المفضلة:', err); } } // 16. إعداد تحديثات الوقت الفعلي (Realtime Subscriptions) function setupRealtimeSubscriptions() { supabaseClient .channel('public:posts') .on('postgres_changes', { event: '*', schema: 'public', table: 'posts' }, async () => { await fetchPosts(); }) .subscribe(); } // 17. تحديث واجهة الحساب والبروفايل function updateProfileUI() { if (!currentUser) return; document.getElementById('profile-name').textContent = currentUser.name; document.getElementById('profile-phone').textContent = currentUser.city || 'بغداد'; } else { myPosts = allPosts.filter(p => savedPostIds.has(p.id)); } document.getElementById('stat-my-posts-count').textContent = allPosts.filter(p => p.user_id === currentUser?.id).length; document.getElementById('stat-saved-count').textContent = savedPostIds.size; document.getElementById('profile-items-count').textContent = `${myPosts.length} عنصر`; if (myPosts.length === 0) { container.innerHTML = `
لا توجد إعلانات لعرضها هنا.
`; return; } container.innerHTML = myPosts.map(post => `

${post.title}

${post.price} ${post.currency || '$'}
${post.user_id === currentUser?.id ? ` ` : ''}
`).join(''); } function switchProfileTab(tab) { activeProfileTab = tab; const myBtn = document.getElementById('tab-btn-my-posts'); const savedBtn = document.getElementById('tab-btn-saved'); if (tab === 'my-posts') { myBtn.className = 'flex-1 py-2 bg-brand-500 text-white rounded-xl font-bold text-xs flex items-center justify-center gap-1.5 btn-tap shadow-sm transition'; } else { savedBtn.className = 'flex-1 py-2 bg-brand-500 text-white rounded-xl font-bold text-xs flex items-center justify-center gap-1.5 btn-tap shadow-sm transition'; myBtn.className = 'flex-1 py-2 bg-gray-100 dark:bg-gray-700 text-gray-700 dark:text-gray-300 rounded-xl font-bold text-xs flex items-center justify-center gap-1.5 btn-tap transition'; document.getElementById('profile-tab-title').innerHTML = ` الإعلانات المفضلة`; } renderProfileView(); } function confirmDeletePost(postId) { document.getElementById('delete-target-post-id').value = postId; toggleModal('delete-confirm-modal'); } async function executeConfirmDelete() { const postId = document.getElementById('delete-target-post-id').value; if (!postId) return; try { await supabaseClient.from('posts').delete().eq('id', postId); toggleModal('comments-modal'); } function openInboxModal() { alert('خاصية المحادثات المباشرة قيد التحديث في الإصدار الحالي.'); } function editProfileModal() { const newName = prompt('أدخل الاسم الجديد:', currentUser?.name || '');
